$base = "http://localhost:8080/api/policies"

function Invoke-PolicyApi {
    param(
        [string]$Method,
        [string]$Uri,
        [string]$Body = $null
    )

    try {
        if ($Body) {
            $response = Invoke-WebRequest `
                -Method $Method `
                -Uri $Uri `
                -ContentType "application/json" `
                -Body $Body
        }
        else {
            $response = Invoke-WebRequest `
                -Method $Method `
                -Uri $Uri
        }

        Write-Host "`nHTTP Status:" $response.StatusCode $response.StatusDescription -ForegroundColor Green

        if ($response.Content) {
            $response.Content | ConvertFrom-Json | Format-List
        }
    }
    catch {
        $statusCode = [int]$_.Exception.Response.StatusCode
        $statusDescription = $_.Exception.Response.StatusDescription

        Write-Host "`nHTTP Status:" $statusCode $statusDescription -ForegroundColor Red

        if ($_.ErrorDetails.Message) {
            try {
                $_.ErrorDetails.Message | ConvertFrom-Json | Format-List
            }
            catch {
                Write-Host $_.ErrorDetails.Message
            }
        }
    }
}
